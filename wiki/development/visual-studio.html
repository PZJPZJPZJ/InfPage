Code Server

  • code-server-Github
Docker Compose
services:
  code-server:
    image: codercom/code-server:latest
    container_name: code-server
    ports:
      - 8080:8080
    environment:
      - PASSWORD=your-login-password
    volumes:
      - ./local:/home/coder/.local
      - ./config:/home/coder/.config
      - ./home:/home/coder/project
    restart: unless-stopped
